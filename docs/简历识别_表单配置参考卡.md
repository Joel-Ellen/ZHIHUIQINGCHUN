## 四、前台脚本（完整，直接粘贴到脚本编辑器）

```javascript
/**
 * 智绘青春 - 简历识别页面脚本
 * 绑定页面: zc_resume_recognition
 *
 * 交互清单:
 *  ① 附件上传 → 启用AI按钮 + 自动填标题
 *  ② 类型切换 → quiz隐藏附件 / 其余显示
 *  ③ AI分析 → 调Agent异步分析
 *  ④ 导出PDF → 预留
 */
export default {

    /* ============================================================
     * 生命周期：页面初始化
     * ============================================================ */
    didMount() {
        // ① AI分析按钮初始禁用
        this.$api.setEnable(false, 'btnAIAnalyze');

        // ② 监听附件上传完成
        this.$api.on('attachmentpanelap1.afterUpload',
            this._onAttachmentUploaded.bind(this));

        // ③ 监听记录类型变更
        this.$api.on('fk_record_type.onValueChange',
            this._onRecordTypeChanged.bind(this));

        // ④ 初始化附件面板状态
        this._syncAttachmentPanelState();
    },

    /* ============================================================
     * 交互①：附件上传完成
     * ============================================================ */
    _onAttachmentUploaded(context, params) {
        // 启用AI按钮
        this.$api.setEnable(true, 'btnAIAnalyze');

        // 提取文件列表（兼容多版本苍穹参数格式）
        const files = (params && params.files) ? params.files
            : (Array.isArray(params) ? params : []);

        if (files.length > 0) {
            const firstName = files[0].fileName || files[0].name || '';
            if (firstName) {
                // 写文件名到分录第0行
                this.$api.setModelValue('fk_filename_snapshot', firstName, 0);

                // 若标题为空，自动填充
                const title = this.$api.getModelValue('fk_title');
                if (!title || title.trim() === '') {
                    const clean = firstName.replace(/\.(pdf|docx|doc)$/i, '');
                    this.$api.setModelValue('fk_title', clean + ' - 分析报告');
                }
            }
        }
    },

    /* ============================================================
     * 交互②：记录类型变更
     * ============================================================ */
    _onRecordTypeChanged(event) {
        const val = (event && event.value !== undefined) ? event.value : event;

        if (val === 'quiz') {
            // 测评 → 隐藏附件
            this.$api.setVisible(false, 'attachmentpanelap1');
            this.$api.setEnable(false, 'btnAIAnalyze');
        } else {
            // 分析/报告 → 显示附件
            this.$api.setVisible(true, 'attachmentpanelap1');
            const snap = this.$api.getModelValue('fk_filename_snapshot', 0);
            if (snap) {
                this.$api.setEnable(true, 'btnAIAnalyze');
            }
        }
    },

    /* ============================================================
     * 交互③：AI分析按钮点击 → 触发Agent任务流
     * ============================================================ */
    'btnAIAnalyze.onClick'(event) {
        const recordType = this.$api.getModelValue('fk_record_type');
        const title      = this.$api.getModelValue('fk_title');
        const userId     = this.$api.getModelValue('fk_user_id');
        const files      = this._getFiles();

        // 前置校验
        if (files.length === 0 && recordType !== 'quiz') {
            this.$api.showMessage('warning',
                '请先上传简历文件（支持 PDF、Word 格式）');
            return;
        }

        // 进入加载态
        this.$api.setEnable(false, 'btnAIAnalyze');
        this.$api.showLoading('AI 正在深度分析中，预计 30 秒 ~ 2 分钟...');

        // 构建Agent参数
        const input = {
            recordType: recordType,
            title:      title || '未命名分析报告',
            fileUrls:   files.map(f => f.url || f.fileUrl || ''),
            fileName:   files.length ? (files[0].fileName || files[0].name || '') : '',
            userId:     userId,
            currentFid: this.$api.getModelValue('fid')
        };

        // 调用Agent API（异步）
        fetch('/kapi/v2/zhihuiqingchun/custom_agent_analyze', {
            method:  'POST',
            headers: { 'Content-Type': 'application/json' },
            body:    JSON.stringify(input)
        })
        .then(r => {
            if (!r.ok) throw new Error('HTTP ' + r.status);
            return r.json();
        })
        .then(data => this._onSuccess(data))
        .catch(err => this._onError(err));
    },

    _onSuccess(result) {
        this.$api.hideLoading();
        this.$api.setEnable(true, 'btnAIAnalyze');

        // 刷新左侧历史树
        try {
            const tree = this.$api.getControl('historyTreeView');
            if (tree && tree.refresh) tree.refresh();
        } catch (e) {}

        const id = result.recordId || '';
        const discovered = (result.discoveredSkills || []).length;
        let msg = '✅ AI 分析完成！报告已保存';
        if (id) msg += '，记录编号: ' + id;
        if (discovered > 0) msg += '\n🔍 图谱发现 ' + discovered + ' 个隐性潜在技能';
        this.$api.showMessage('success', msg);
    },

    _onError(error) {
        this.$api.hideLoading();
        this.$api.setEnable(true, 'btnAIAnalyze');
        const msg = error && error.message ? error.message : '未知错误';
        this.$api.showMessage('error', '❌ AI 分析失败: ' + msg);
    },

    /* ============================================================
     * 交互④：导出PDF（预留）
     * ============================================================ */
    'btnExportPDF.onClick'(event) {
        this.$api.showMessage('info', '📄 PDF 导出功能开发中，敬请期待！');
    },

    /* ============================================================
     * 辅助方法
     * ============================================================ */
    _getFiles() {
        try {
            const panel = this.$api.getControl('attachmentpanelap1');
            if (panel && typeof panel.getFiles === 'function') {
                return panel.getFiles() || [];
            }
        } catch (e) {}
        return [];
    },

    _syncAttachmentPanelState() {
        const type = this.$api.getModelValue('fk_record_type');
        if (type) this._onRecordTypeChanged(type);
    }
};
```

## 五、Agent 任务流对接映射

```
  ┌── 表单控件 ──────────────┬── Agent Step ──────────────┬── 后端处理 ──┐
  │                          │                            │              │
  │ attachmentpanelap1       │ Step 1: 文档解析            │ 金蝶复杂文档  │
  │ (上传PDF/Word)           │ → 提取简历关键文本          │ 解析V2        │
  │                          │                            │              │
  │ fk_record_type           │ Step 3: 十维画像匹配        │ Agent LLM     │
  │ (analysis/report/quiz)   │ → 决定检索策略              │ (qwen-plus)   │
  │                          │                            │              │
  │ fk_title                 │ Step 4: 生成报告            │ JSON 输出     │
  │ (报告标题)               │ → 注入报告标题              │              │
  │                          │                            │              │
  │ fk_content               │ Step 4→5: 报告持久化        │ CustomAgent   │
  │ (AI分析内容大文本)        │ → Agent输出JSON             │ CallbackPlugin│
  │                          │ → 图谱二阶强化              │ → PostgreSQL │
  │                          │                            │              │
  │ fk_user_id               │ Step 5: 异步保存            │ SaveService   │
  │ (所属用户)               │ → 关联用户记录              │ Helper        │
  │                          │                            │              │
  │ fk_filename_snapshot     │ Step 1→5: 文件追踪          │ 分录表存储    │
  │ (文件名快照)             │ → 快照存于分录              │              │
  └──────────────────────────┴────────────────────────────┴──────────────┘
```
